### Pratica 5.1 - Integridade por SHA-256
### Inicio: 19/11/2026 14:38:28 UTC | 11:38:28 BRT
### Sistema: Linux | sha256sum (GNU coreutils) 9.4

lab$ printf '%s\n' 'Relato simulado' > evidencia.txt
lab$ sha256sum evidencia.txt
969ff386f5e35857cddf842e11ed923311f24f0da48ef8ed9ca6f362e7f0627e  evidencia.txt
lab$ cp evidencia.txt evidencia_copia.txt
lab$ sha256sum evidencia_copia.txt   # antes de editar
969ff386f5e35857cddf842e11ed923311f24f0da48ef8ed9ca6f362e7f0627e  evidencia_copia.txt
lab$ sed -i 's/simulado/alterado/' evidencia_copia.txt   # altera UMA palavra apenas na copia
lab$ cat evidencia.txt evidencia_copia.txt
Relato simulado
Relato alterado
lab$ sha256sum evidencia.txt evidencia_copia.txt
969ff386f5e35857cddf842e11ed923311f24f0da48ef8ed9ca6f362e7f0627e  evidencia.txt
